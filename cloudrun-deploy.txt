# 1) Inspect effective org policies on the project
gcloud resource-manager org-policies describe constraints/iam.allowedPolicyMemberDomains --effective --project=ca-panel-001
gcloud resource-manager org-policies describe constraints/run.managed.requireInvokerIam --effective --project=ca-panel-001

# 2) Get project number (used in policy names)
$PROJECT_NUMBER = gcloud projects describe ca-panel-001 --format="value(projectNumber)"

# 3) Disable "Require authentication for Cloud Run invocations" at project level
@"
name: projects/$PROJECT_NUMBER/policies/run.managed.requireInvokerIam
spec:
  rules:
  - enforce: false
"@ | Set-Content .\run-require-invokeriam-false.yaml

gcloud resource-manager org-policies set-policy .\run-require-invokeriam-false.yaml --project=ca-panel-001

# 4) Override domain-restricted-sharing for this project (allow public principals like allUsers)
@"
name: projects/$PROJECT_NUMBER/policies/iam.allowedPolicyMemberDomains
spec:
  inheritFromParent: false
  rules:
  - allowAll: true
"@ | Set-Content .\iam-allow-allusers.yaml

gcloud resource-manager org-policies set-policy .\iam-allow-allusers.yaml --project=ca-panel-001

# 5) Make Cloud Run service public without allUsers IAM binding
gcloud run services update capanel-backend --project ca-panel-001 --region us-west1 --no-invoker-iam-check

# (Alternative after policy allows allUsers)
gcloud run services add-iam-policy-binding capanel-backend --project ca-panel-001 --region us-west1 --member="allUsers" --role="roles/run.invoker"
